// ---------------------------------------------------------------------------------------------
// LeakyChartScreen.cs — THE ANTI-PATTERN. Kept as .cs.txt so it never compiles, and read at run
// time by the "Anti-pattern: leaky screen" button of the WorkOrderHistoryPage.
//
// This is the screen the walkthrough opens with: "copy-paste panels & leaky widgets". It is the
// version of the work-order history page that a team writes before anyone owns a component:
//
//   * it rebuilds the timeline inline, so the fifth screen that needs a timeline gets a fifth
//     slightly different layout and a sixth set of colours;
//   * it registers the vendor packages itself, so a screen that forgets one ships a broken chart;
//   * it pastes the client adapter into the screen as a string, so upgrading the vendor means
//     finding every screen that has a copy;
//   * it edits the raw Options object, so the vendor's option names are now application code;
//   * it handles the raw WidgetEvent and switches on strings, so nothing is typed and nothing is
//     validated;
//   * it drives the browser with Application.Eval, so the server has no idea what happened;
//   * and it has no failure path at all: when the vendor script is blocked the card is simply
//     blank, and the screen looks broken.
//
// Every line below is legal Wisej.NET. That is exactly why the review question is
// "can another developer use the component without reading its internals?" and not
// "does the screen work?".
// ---------------------------------------------------------------------------------------------

using System;
using System.Linq;
using Wisej.Web;
using EnterpriseOps.Services;

namespace EnterpriseOps.UI
{
    public partial class LeakyChartScreen : Page
    {
        private readonly WorkOrderHistoryService _service;
        private readonly SessionContext _session;
        private Widget _chart;

        public LeakyChartScreen(SessionContext session, WorkOrderHistoryService service)
        {
            InitializeComponent();
            _session = session;
            _service = service;
            BuildChart();
        }

        // ---- the widget, wired by hand on this screen (and on four other screens) --------------

        private void BuildChart()
        {
            _chart = new Widget
            {
                Location = new System.Drawing.Point(456, 88),
                Size = new System.Drawing.Size(420, 210),
                Name = "chartWorkOrders",
            };

            // Every screen registers the packages itself. Miss one, or get the order wrong, and the
            // stylesheet loses to the vendor's rules — on that screen only.
            _chart.Packages.Add(new Widget.Package { Name = "eops-chart-vendor", Source = "Widgets/vendor-opschart.js" });
            _chart.Packages.Add(new Widget.Package { Name = "eops-chart-style", Source = "Widgets/opschart.css" });

            // The adapter, as a string, in a screen. There are four more copies of this in the solution
            // and two of them are still on vendor 1.0.
            _chart.InitScript =
                "this.init = function (options) {" +
                "  var host = document.createElement('div');" +
                "  host.style.cssText = 'position:absolute;left:0;top:0;right:0;bottom:0;';" +
                "  this.container.innerHTML = '';" +
                "  this.container.appendChild(host);" +
                "  this.widget = new EnterpriseOpsChart(host, options);" +   // no try/catch: blocked script = blank card
                "  var me = this;" +
                "  this.widget.on('pointclick', function (e) { me.fireWidgetEvent('pointclick', e); });" +
                "};" +
                "this.update = function (options, old) { this.widget.setOptions(options); };";

            _chart.WiredEvents = new[] { "pointclick" };

            // The vendor's option names are now application code. Rename one in vendor 2.0 and every
            // screen that spelled it out has to be found and edited.
            dynamic options = _chart.Options;
            options.palette = "ops";
            options.caption = "WORK-ORDER HISTORY";
            options.showLegend = true;
            options.segments = new object[0];

            _chart.WidgetEvent += chart_WidgetEvent;

            this.Controls.Add(_chart);
        }

        // And the screen carries its own copy of the vendor's option shape, so renaming a field in
        // vendor 2.0 breaks a class that lives in a screen and has nothing to do with charts.
        private class EnterpriseOpsChartOptions
        {
            public object[] segments;
            public string palette;
            public string caption;
            public bool showLegend;
        }

        private EnterpriseOpsChartOptions BuildVendorOptions(StatusCountView[] counts)
        {
            return new EnterpriseOpsChartOptions
            {
                segments = counts.Select(c => (object)new { key = c.Key, label = c.Label, value = c.Count }).ToArray(),
                palette = "ops",
                caption = "WORK-ORDER HISTORY",
                showLegend = true,
            };
        }

        // ---- the raw event: a string switch, an untyped payload, no validation -----------------

        private async void chart_WidgetEvent(object sender, WidgetEventArgs e)
        {
            if (e.Type != "pointclick")
                return;

            dynamic data = e.Data;
            string key = (string)data.key;              // straight from the browser, straight into the query

            var rows = await _service.GetWorkOrdersByGroupAsync(_session, key);
            dgvSegment.DataSource = rows.Value.Rows;

            // And now tell the browser what to highlight, by poking it directly.
            Application.Eval("App.MainPage.chartWorkOrders.widget.select('" + key + "');");
        }

        // ---- the timeline, rebuilt inline for the fifth time --------------------------------

        private void ShowHistory(HistoryEntryView[] entries)
        {
            pnlTimeline.Controls.Clear();

            int y = 0;
            foreach (var entry in entries)
            {
                var row = new Panel { Location = new System.Drawing.Point(0, y), Size = new System.Drawing.Size(372, 26) };

                var dot = new Label
                {
                    Text = "●",
                    // The colours, again. The audit screen used a slightly different red, and nobody knows
                    // which one is the right one any more.
                    ForeColor = entry.Status.ToString() == "Escalated"
                        ? System.Drawing.Color.FromArgb(192, 57, 43)
                        : System.Drawing.Color.FromArgb(138, 151, 164),
                    Location = new System.Drawing.Point(0, 4),
                    Size = new System.Drawing.Size(14, 18),
                };

                var when = new Label
                {
                    Text = entry.AtUtc.ToString("MMM dd, HH:mm"),
                    Location = new System.Drawing.Point(16, 4),
                    Size = new System.Drawing.Size(104, 18),
                };

                var status = new Label
                {
                    Text = entry.StatusLabel,
                    Location = new System.Drawing.Point(124, 4),
                    Size = new System.Drawing.Size(78, 18),
                };

                var message = new Label
                {
                    Text = entry.Note,
                    Location = new System.Drawing.Point(206, 4),
                    Size = new System.Drawing.Size(164, 18),
                };

                row.Controls.Add(dot);
                row.Controls.Add(when);
                row.Controls.Add(status);
                row.Controls.Add(message);
                pnlTimeline.Controls.Add(row);

                y += 28;
            }
        }
    }
}
